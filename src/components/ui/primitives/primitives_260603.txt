{preview && (
  <script>
    class LinkPreview {
      private timer: number | null = null;
      private previewEl: HTMLElement | null = null;

      constructor() {
        this.init();
      }

      private init() {
        const links = document.querySelectorAll('a[data-preview="true"]');
        links.forEach(link => {
          link.addEventListener('mouseenter', (e) => this.handleMouseEnter(e as MouseEvent));
          link.addEventListener('mouseleave', () => this.handleMouseLeave());
        });
      }

      private handleMouseEnter(e: MouseEvent) {
        const target = e.currentTarget as HTMLAnchorElement;
        const href = target.href;

        this.timer = window.setTimeout(() => {
          this.showPreview(target, href);
        }, 400);
      }

      private handleMouseLeave() {
        if (this.timer) {
          clearTimeout(this.timer);
          this.timer = null;
        }
        this.hidePreview();
      }

      private async showPreview(anchor: HTMLElement, href: string) {
        if (this.previewEl) this.hidePreview();

        this.previewEl = document.createElement('div');
        this.previewEl.className = 'link-preview-card';
        this.previewEl.innerHTML = `
          <div class="link-preview-loading">
            <p>Fetching metadata...</p>
          </div>
        `;

        document.body.appendChild(this.previewEl);

        const rect = anchor.getBoundingClientRect();
        this.previewEl.style.position = 'fixed';
        this.previewEl.style.top = `${rect.bottom + 10}px`;
        this.previewEl.style.left = `${rect.left}px`;
        this.previewEl.style.zIndex = '1000';

        setTimeout(() => {
          if (this.previewEl) {
            try {
              const url = new URL(href);
              this.previewEl.innerHTML = `
                <div class="link-preview-content">
                  <p class="link-preview-title">Link Preview: ${url.hostname}</p>
                  <p class="link-preview-url">${href}</p>
                  <p class="link-preview-desc">Metadata proxy required for full OG data.</p>
                </div>
              `;
            } catch (e) {
               this.previewEl.innerHTML = '<p>Invalid URL</p>';
            }
          }
        }, 600);
      }

      private hidePreview() {
        if (this.previewEl) {
          this.previewEl.remove();
          this.previewEl = null;
        }
      }
    }

    document.addEventListener('astro:page-load', () => {
      new LinkPreview();
    });
  </script>
)}

<script>
  class ComponentPreviewElement extends HTMLElement {
    connectedCallback() {
      const previewId = this.dataset.id;
      if (!previewId) return;
      
      const container = this.querySelector(`#${previewId}`);
      if (!container) return;
      
      const tabButtons = container.querySelectorAll<HTMLButtonElement>('.tab-btn');
      const tabPanes = container.querySelectorAll<HTMLDivElement>('.tab-pane');
      
      tabButtons.forEach(button => {
        button.addEventListener('click', () => {
          const tab = button.getAttribute('data-tab');
          
          tabButtons.forEach(btn => {
            btn.classList.remove('is-active');
            btn.setAttribute('aria-selected', 'false');
          });
          
          tabPanes.forEach(pane => {
            pane.classList.remove('is-active');
          });
          
          button.classList.add('is-active');
          button.setAttribute('aria-selected', 'true');
          
          const activePane = container.querySelector<HTMLDivElement>(`.tab-pane[data-pane="${tab}"]`);
          if (activePane) {
            activePane.classList.add('is-active');
          }
        });
      });
    }
  }
  
  if (!customElements.get('component-preview-element')) {
    customElements.define('component-preview-element', ComponentPreviewElement);
  }
</script>
