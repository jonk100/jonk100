/**
 * src/components/ui/input/Input.astro
 * An accessible form input component with labels, errors, and hints.
 * - Supports various input types, sizes, and styling variants
 * - Handles auto-sizing for content (modern browser support)
 * - Visual container, label, and error layout styles defined in src/styles/variants/controls.css
 * @props (local): label, placeholder, error, hint, disabled, required, theme, size, id, name, type
 * @props (global): class, class:list
 * - theme: Determines the visual style from predefined options
 * - size: Adjusts the input dimensions
 * - error: Displays an error message and sets aria-invalid
 * - hint: Displays a helpful text underneath the input
 */
import type { ComponentTheme, Sizes } from 'src/types/variants.types';

export interface Props {
  label?: string;
  placeholder?: string;
  error?: string;
  hint?: string;
  disabled?: boolean;
  required?: boolean;
  theme?: ComponentTheme;
  size?: Sizes;
  id?: string;
  name?: string;
  type?: 'text' | 'email' | 'password' | 'number' | 'search' | 'tel' | 'url';
  class?: string;
  "class:list"?: Record<string, boolean> | any[];
  [key: string]: any;
}

const {
  label,
  placeholder,
  error,
  hint,
  disabled = false,
  required = false,
  theme = 'default',
  size = 'md',
  id = `input-${Math.random().toString(36).substring(2, 9)}`,
  name,
  type = 'text',
  class: className = '',
  "class:list": classList = {},
  ...rest
} = Astro.props;
---

<div class:list={['input-wrapper', className, classList]}>
  {label && (
    <label for={id} class="input-label">
      {label} {required && <span class="required">*</span>}
    </label>
  )}
  
  <input
    id={id}
    name={name}
    type={type}
    placeholder={placeholder}
    disabled={disabled}
    required={required}
    aria-invalid={!!error}
    aria-describedby={error ? `${id}-error` : hint ? `${id}-hint` : undefined}
    class:list={[
      'control',
      `variant--${theme}`,
      `control--${size}`,
      'input-field',
      { 'input-field--error': error }
    ]}
    {...rest}
  />
  
  {error && <span id={`${id}-error`} class="input-error">{error}</span>}
  {hint && !error && <span id={`${id}-hint`} class="input-hint">{hint}</span>}
</div>

<style>
  .input-field {
    /* Override specific .control styles that don't apply to inputs */
    justify-content: flex-start;
    text-align: left;
    cursor: text;
    width: 100%;
    
    /* Modern autosizing support (supported in Chrome 121+) */
    field-sizing: content;
    min-width: 100%; /* Fallback/Constraint */
  }

  .input-field:focus-visible {
    outline: 2px solid var(--accent);
    outline-offset: -1px;
  }
</style>

<script define:vars={{ id, options, initialSelected }}>
  class MultiselectController {
    constructor() {
      this.selected = new Set(initialSelected);
      this.filter = '';
      this.root = document.getElementById(`${id}-root`);
      this.trigger = document.getElementById(`${id}-trigger`);
      this.input = document.getElementById(`${id}-input`);
      this.dropdown = document.getElementById(`${id}-dropdown`);
      this.optionsContainer = document.getElementById(`${id}-options-container`);
      this.chipsContainer = document.getElementById(`${id}-chips`);
      this.hiddenSelect = document.getElementById(`${id}-hidden`);
      this.emptyState = document.getElementById(`${id}-empty`);
      
      if (!this.root || !this.input) return;
      this.init();
    }

    init() {
      this.renderChips();
      this.renderOptions();

      this.trigger.addEventListener('click', (e) => {
        // Don't toggle if clicking a chip remove button
        if ((e.target as HTMLElement).closest('.ms-chip-remove')) return;
        this.toggleDropdown();
      });

      this.input.addEventListener('input', () => {
        this.filter = this.input.value;
        this.renderOptions();
        this.openDropdown();
      });

      this.input.addEventListener('keydown', (e) => {
        if (e.key === 'Escape') this.closeDropdown();
        if (e.key === 'Backspace' && this.input.value === '') {
          // Remove last chip
          const items = Array.from(this.selected);
          if (items.length > 0) {
            this.toggleOption(items[items.length - 1]);
          }
        }
      });

      document.addEventListener('click', (e) => {
        if (!this.root.contains(e.target)) this.closeDropdown();
      });
      
      // Export functions to global scope for inline event handlers generated by renderOptions
      window[`msToggle_${id}`] = (val) => this.toggleOption(val);
      window[`msRemove_${id}`] = (e, val) => {
        e.stopPropagation();
        this.toggleOption(val);
      };
    }

    renderChips() {
      this.chipsContainer.innerHTML = '';
      this.selected.forEach(val => {
        const opt = options.find(o => o.value === val);
        if (!opt) return;
        
        const chip = document.createElement('span');
        chip.className = 'ms-chip';
        chip.innerHTML = `
          ${opt.label}
          <button type="button" class="ms-chip-remove" aria-label="Remove ${opt.label}" onclick="window['msRemove_${id}'](event, '${val}')">&times;</button>
        `;
        this.chipsContainer.appendChild(chip);
      });
      this.updateHiddenSelect();
    }

    renderOptions() {
      const filtered = options.filter(o => o.label.toLowerCase().includes(this.filter.toLowerCase()));
      
      if (filtered.length === 0) {
        this.optionsContainer.innerHTML = '';
        this.emptyState.style.display = 'block';
        return;
      }
      
      this.emptyState.style.display = 'none';
      this.optionsContainer.innerHTML = filtered.map(o => {
        const isSel = this.selected.has(o.value);
        return `
          <div class="ms-option ${isSel ? 'selected' : ''}" role="option" aria-selected="${isSel}" tabindex="0" onclick="window['msToggle_${id}']('${o.value}')" onkeydown="if(event.key==='Enter') window['msToggle_${id}']('${o.value}')">
            <span class="ms-check">${isSel ? '✓' : ''}</span>
            <span>${o.label}</span>
          </div>
        `;
      }).join('');
    }

    toggleOption(val) {
      if (this.selected.has(val)) {
        this.selected.delete(val);
      } else {
        this.selected.add(val);
      }
      this.filter = '';
      this.input.value = '';
      this.renderChips();
      this.renderOptions();
      this.input.focus();
    }

    updateHiddenSelect() {
      if (!this.hiddenSelect) return;
      Array.from(this.hiddenSelect.options).forEach(opt => {
        opt.selected = this.selected.has(opt.value);
      });
    }

    toggleDropdown() {
      const isOpen = this.dropdown.classList.contains('open');
      if (isOpen) this.closeDropdown();
      else this.openDropdown();
    }

    openDropdown() {
      this.dropdown.classList.add('open');
      this.trigger.setAttribute('aria-expanded', 'true');
      this.input.focus();
    }

    closeDropdown() {
      this.dropdown.classList.remove('open');
      this.trigger.setAttribute('aria-expanded', 'false');
    }
  }

  // Use simple timeout to ensure DOM is ready in Astro
  setTimeout(() => {
    new MultiselectController();
  }, 0);
